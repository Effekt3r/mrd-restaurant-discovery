# Milestone 1: project foundation, implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A ten-module Android project that builds, tests and produces an R8-minified release. An edge-to-edge app shell navigates between Browse and Details placeholders. The build and the architecture tests both enforce the module boundaries, CI gates every pull request, and the README lists what's needed to run it.

**Architecture:**
- An included `build-logic` build provides convention plugins (`mrd.*`), so each module's build file only states what's unique to it.
- The version catalog pins a proven toolchain.
- `:app` is a thin host that connects the two feature modules through Navigation Compose callbacks.
- Boundaries are enforced twice: the feature convention plugin fails the build on a feature-to-feature dependency, and Konsist tests fail on forbidden imports and patterns.

**Tech stack:** Gradle 9.7.1 (Kotlin DSL, version catalog, configuration cache), AGP 9.3.3 with built-in Kotlin, Kotlin 2.3.21, KSP 2.3.9, Hilt 2.59.2, Jetpack Compose (BOM 2026.06.01), Navigation Compose 2.9.8, Konsist 0.17.3, Spotless 8.3.0 with ktlint 1.4.0, GitHub Actions.

**Spec:** `docs/specs/2026-11-18-restaurant-discovery-design.md`, §4.1–4.3, §7.2, §8.1–8.4, §9 and §10. **Roadmap:** `docs/plans/2026-11-18-00-roadmap.md`.

**Branch:** `chore/project-foundation`, cut from `dev`. It ends in one pull request into `dev`. Run every command from the repository root.

## Before you start

- **JDK 21:** on the `PATH` or in `JAVA_HOME` (`export JAVA_HOME=$(/usr/libexec/java_home -v 21)` on macOS).
- **Android SDK:** platform 36 and build-tools 36. Either `ANDROID_HOME` is set, or a git-ignored `local.properties` contains `sdk.dir=<path to the SDK>`.
- **Gradle:** any local install, used once to generate the wrapper.
- **Versions:** don't change any version in this plan. They were validated together by a spike build on 2026-11-18 (see the roadmap's version policy).
- **Formatting:** before every commit, run `./gradlew spotlessApply`. The commit step lists the files to stage; also stage any of them that Spotless reformatted.

import org.gradle.api.Project

internal const val COMPILE_SDK = 36
internal const val TARGET_SDK = 36

internal const val MIN_SDK = 26

/** SDK levels and Java/Kotlin targets shared by every Android module. AGP 9 provides Kotlin built in. */
internal fun Project.configureKotlinAndroid(commonExtension: CommonExtension) {

        compileOptions.targetCompatibility = JavaVersion.VERSION_17

        compilerOptions.jvmTarget.set(JvmTarget.JVM_17)
    }
}

/** Java/Kotlin targets for pure Kotlin modules. */

    extensions.configure<JavaPluginExtension> {
        sourceCompatibility = JavaVersion.VERSION_17
        targetCompatibility = JavaVersion.VERSION_17
    }
    extensions.configure<KotlinJvmProjectExtension> {
        compilerOptions.jvmTarget.set(JvmTarget.JVM_17)
    }
}

import com.android.build.api.dsl.ApplicationExtension
import com.effekt3r.mrd.buildlogic.TARGET_SDK
import com.effekt3r.mrd.buildlogic.configureKotlinAndroid
import org.gradle.api.Plugin
import org.gradle.api.Project
import org.gradle.kotlin.dsl.apply
import org.gradle.kotlin.dsl.configure

/** The application module: SDK levels plus the lint policy for the whole app (it checks dependencies too). */
class AndroidApplicationConventionPlugin : Plugin<Project> {
    override fun apply(target: Project) {
        with(target) {
            apply(plugin = "com.android.application")
            extensions.configure<ApplicationExtension> {
                configureKotlinAndroid(this)
                defaultConfig.targetSdk = TARGET_SDK
                lint {
                    checkDependencies = true
                    abortOnError = true
                    // Versions are pinned on purpose (README: toolchain and version policy).
                    // Dependabot proposes updates once a release is 30 days old.
                    disable += setOf("GradleDependency", "AndroidGradlePluginVersion", "NewerVersionAvailable")
                }
            }
        }
    }
}

import com.android.build.api.dsl.ApplicationExtension
import com.android.build.api.dsl.CommonExtension
import com.android.build.api.dsl.LibraryExtension
import com.effekt3r.mrd.buildlogic.libs
import org.gradle.api.Plugin
import org.gradle.api.Project
import org.gradle.kotlin.dsl.apply
import org.gradle.kotlin.dsl.configure
import org.gradle.kotlin.dsl.dependencies
import org.gradle.kotlin.dsl.getByType
import org.jetbrains.kotlin.compose.compiler.gradle.ComposeCompilerGradlePluginExtension

/** Compose for an Android application or library module, with the shared stability configuration. */
class AndroidComposeConventionPlugin : Plugin<Project> {
    override fun apply(target: Project) {
        with(target) {
            apply(plugin = "org.jetbrains.kotlin.plugin.compose")
            pluginManager.withPlugin("com.android.application") {
                configureCompose(extensions.getByType<ApplicationExtension>())
            }
            pluginManager.withPlugin("com.android.library") {
                configureCompose(extensions.getByType<LibraryExtension>())
            }
            extensions.configure<ComposeCompilerGradlePluginExtension> {
                @Suppress("UnstableApiUsage")
                stabilityConfigurationFiles.add(isolated.rootProject.projectDirectory.file("compose-stability.conf"))
            }
        }
    }

    private fun Project.configureCompose(extension: CommonExtension) {
        extension.buildFeatures.compose = true
        dependencies {
            val bom = libs.findLibrary("androidx-compose-bom").get()
            "implementation"(platform(bom))
            "testImplementation"(platform(bom))
            "implementation"(libs.findLibrary("androidx-compose-ui-tooling-preview").get())
            "debugImplementation"(libs.findLibrary("androidx-compose-ui-tooling").get())
        }
    }
}
```

import com.effekt3r.mrd.buildlogic.libs
import org.gradle.api.Plugin
import org.gradle.api.Project
import org.gradle.api.artifacts.ProjectDependency
import org.gradle.kotlin.dsl.apply

import org.gradle.kotlin.dsl.project
import org.gradle.kotlin.dsl.withType

/**
 * A feature module: an Android library with Compose, Hilt, type-safe navigation routes and the shared
 * design system. Features never depend on each other; :app connects them through callbacks.
 */
class AndroidFeatureConventionPlugin : Plugin<Project> {
    override fun apply(target: Project) {
        with(target) {
            apply(plugin = "mrd.android.library")
            apply(plugin = "mrd.android.compose")
            apply(plugin = "mrd.hilt")
            apply(plugin = "org.jetbrains.kotlin.plugin.serialization")
            dependencies {
                "implementation"(project(":core:designsystem"))
                "implementation"(project(":core:ui"))
                "implementation"(libs.findLibrary("androidx-lifecycle-runtime-compose").get())
                "implementation"(libs.findLibrary("androidx-lifecycle-viewmodel-compose").get())
                "implementation"(libs.findLibrary("androidx-hilt-lifecycle-viewmodel-compose").get())
                "implementation"(libs.findLibrary("androidx-navigation-compose").get())
                "implementation"(libs.findLibrary("kotlinx-serialization-core").get())
            }
            enforceFeatureIsolation()
        }
    }

    // Fails the build as soon as a feature declares a dependency on another feature. AGP adds a
    // dependency from a module's test configurations back to the module itself, so that one is allowed.
    private fun Project.enforceFeatureIsolation() {
        val self = path
        configurations.configureEach {
            dependencies.withType<ProjectDependency>().configureEach {
                check(path == self || !path.startsWith(":feature:")) {
                    "$self must not depend on $path. Feature modules are independent; connect them in :app."
                }
            }
        }
    }
}
```

// Release signing comes from Gradle properties or environment variables, never from the repository.
// Locally, a release build without them falls back to the debug key (with a warning) so anyone can run
// an R8 build. CI release builds pass -Pmrd.requireReleaseSigning=true and fail instead.
fun signingValue(name: String): String? =
    providers.gradleProperty(name).orElse(providers.environmentVariable(name)).orNull

val releaseKeystore = signingValue("MRD_KEYSTORE_FILE")
val requireReleaseSigning = providers.gradleProperty("mrd.requireReleaseSigning").orNull == "true"

android {
    namespace = "com.effekt3r.mrd"

    defaultConfig {
        applicationId = "com.effekt3r.mrd"
        versionName = providers.gradleProperty("mrd.versionName").get()
        versionCode = providers.gradleProperty("mrd.versionCode").get().toInt()
    }

    signingConfigs {
        if (releaseKeystore != null) {
            create("release") {
                storeFile = file(releaseKeystore)
                storePassword = signingValue("MRD_KEYSTORE_PASSWORD")
                keyAlias = signingValue("MRD_KEY_ALIAS")
                keyPassword = signingValue("MRD_KEY_PASSWORD")
            }
        }
    }

    buildTypes {
        release {
            isMinifyEnabled = true
            isShrinkResources = true
            proguardFiles(getDefaultProguardFile("proguard-android-optimize.txt"), "proguard-rules.pro")
            signingConfig = when {
                releaseKeystore != null -> signingConfigs.getByName("release")
                requireReleaseSigning -> throw GradleException(
                    "Release signing is required, but MRD_KEYSTORE_FILE is not set.",
                )
                else -> {
                    logger.warn("MRD_KEYSTORE_FILE is not set, so release builds are signed with the debug key.")
                    signingConfigs.getByName("debug")
                }
            }
        }
    }
}

    private val sourceFiles = production.files.filter { "/build-logic/" !in it.path }

    @Test
    fun featureModulesNeverImportAnotherFeature() {
        val violations = sourceFiles
            .filter { "/feature/" in it.path }
            .flatMap { file ->
                val ownFeature = file.packagee?.name.orEmpty()
                    .removePrefix("$ROOT_PACKAGE.feature.")
                    .substringBefore('.')
                file.imports
                    .map { it.name }
                    .filter { it.startsWith("$ROOT_PACKAGE.feature.") }
                    .filterNot { it.startsWith("$ROOT_PACKAGE.feature.$ownFeature.") }
                    .map { "${file.path} imports $it" }
            }
        assertNoViolations("Features are connected by :app, never by importing each other.", violations)
    }

    @Test
    fun modelAndCommonModulesHaveNoAndroidDependencies() {
        val violations = sourceFiles
            .filter { "/core/model/" in it.path || "/core/common/" in it.path }
            .flatMap { file ->
                file.imports
                    .map { it.name }
                    .filter { it.startsWith("android.") || it.startsWith("androidx.") }
                    .map { "${file.path} imports $it" }
            }
        assertNoViolations(":core:model and :core:common are plain Kotlin, so their tests run on the JVM.", violations)
    }

    @Test
    fun rawColoursOnlyAppearInTheDesignSystem() {
        val violations = sourceFiles
            .filter { "/core/designsystem/" !in it.path && "Color(0x" in it.text }
            .map { it.path }
        assertNoViolations(
            "Colours come from design-system tokens; raw Color(0x...) values belong in :core:designsystem.",
            violations,
        )
    }

    @Test
    fun moneyIsNeverAFloatingPointNumber() {
        val moneyAsFloat = Regex("""\b\w*(?:[Cc]ents|[Ff]ee|[Pp]rice)\w*\s*:\s*(?:Double|Float)\b""")
        val violations = sourceFiles.flatMap { file ->
            moneyAsFloat.findAll(file.text).map { "${file.path}: ${it.value}" }.toList()
        }
        assertNoViolations("Money is integer cents (Long), never Double or Float.", violations)
    }

    @Test
    fun viewModelsNeverExposeMutableState() {
        val mutableInitialisers = listOf("MutableStateFlow(", "MutableSharedFlow(", "mutableStateOf(")
        val violations = production.classes()
            .withNameEndingWith("ViewModel")
            .flatMap { viewModel ->
                viewModel.properties()
                    .filter { it.hasPublicOrDefaultModifier }
                    .filter { property ->
                        val declaredMutable = property.type?.name?.startsWith("Mutable") == true
                        val inferredMutable = property.type == null && mutableInitialisers.any { it in property.text }
                        declaredMutable || inferredMutable
                    }
                    .map { "${viewModel.path}: ${viewModel.name}.${it.name}" }
            }
        assertNoViolations("ViewModels expose read-only state; changes go through their functions.", violations)
    }

    @Test
    fun repositoryImplementationsAreInternal() {
        val violations = production.classes()
            .withNameEndingWith("Repository")
            .filter { "/core/testing/" !in it.path && !it.hasInternalModifier }
            .map { "${it.path}: ${it.name}" }

            "Repository implementations are internal to :core:data; other modules see interfaces only.",
            violations,
        )

        assertTrue("$rule\n" + violations.joinToString("\n"), violations.isEmpty())
